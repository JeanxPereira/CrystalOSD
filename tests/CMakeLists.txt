# A capture's scene.menus.json (the menus' schema) stands in for its scene.json in the same folder.
function(prefer_menus_scenes VARIABLE)
    set(SCENES ${${VARIABLE}})
    foreach(SCENE ${SCENES})
        if(SCENE MATCHES "/scene\\.menus\\.json$")
            get_filename_component(FOLDER ${SCENE} DIRECTORY)
            list(REMOVE_ITEM SCENES ${FOLDER}/scene.json)
        endif()
    endforeach()
    set(${VARIABLE} ${SCENES} PARENT_SCOPE)
endfunction()

function(add_menus_gate NAME CAPTURE)
    set(DIR ${FIXTURE_ROOT}/hddosd-110U-${CAPTURE})
    if(NOT (EXISTS ${DIR}/scene.json AND EXISTS ${DIR}/passes.json AND EXISTS ${CLOCK_DUMPS}/FNTOSD))
        return()
    endif()
    set(SCENE ${DIR}/scene.json)
    if(EXISTS ${DIR}/scene.menus.json)
        set(SCENE ${DIR}/scene.menus.json)
    endif()
    set(ARGS ${SCENE} ${CMAKE_SOURCE_DIR}/facts/data/rod-mesh.json ${CMAKE_SOURCE_DIR}/References/model/cube-mesh.json ${DIR}/passes.json ${CLOCK_DUMPS}/FNTOSD ${CLOCK_DUMPS}/hddosd.elf ${ARGN})
    set(ACTIVE ${PARITY_ACTIVE} ${NAME})
    set(BUDGET ${CMAKE_SOURCE_DIR}/tools/ParityTool/budgets/hddosd-110U-${CAPTURE}.json)
    if(EXISTS ${CMAKE_SOURCE_DIR}/tools/ParityTool/budgets/hddosd-110U-${CAPTURE}-scene.json)
        set(BUDGET ${CMAKE_SOURCE_DIR}/tools/ParityTool/budgets/hddosd-110U-${CAPTURE}-scene.json)
    endif()
    if(EXISTS ${DIR}/f0/frame.json)
        list(APPEND ARGS --f0 ${DIR}/f0 --out ${CMAKE_BINARY_DIR}/scene-fixture/${CAPTURE})
        add_test(NAME ${NAME}Pixels COMMAND ParityTool ${CMAKE_BINARY_DIR}/scene-fixture/${CAPTURE} ${CMAKE_BINARY_DIR}/parity-report/${CAPTURE}-scene ${CLOCK_BIN_DIR}/shaders ${BUDGET})
        set_tests_properties(${NAME}Pixels PROPERTIES FIXTURES_REQUIRED ${NAME}Fixture)
        list(APPEND ACTIVE ${NAME}Pixels)
    endif()
    add_test(NAME ${NAME} COMMAND MenusParityTest ${ARGS})
    set_tests_properties(${NAME} PROPERTIES FIXTURES_SETUP ${NAME}Fixture TIMEOUT 3600)
    set(PARITY_ACTIVE ${ACTIVE} PARENT_SCOPE)
endfunction()
